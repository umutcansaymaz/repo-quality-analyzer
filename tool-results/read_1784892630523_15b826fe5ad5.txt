   243→
   244→class SecurityReviewEngine:
   245→    """Run the full security review and produce a :class:`SecurityReview`."""
   246→
   247→    def __init__(self, *, skip_bandit: bool = False, skip_detect_secrets: bool = False) -> None:
   248→        self._skip_bandit = skip_bandit
   249→        self._skip_detect_secrets = skip_detect_secrets
   250→
   251→    def review(self, repository: Repository, workspace: Path) -> SecurityReview:
   252→        """Run all security checks and return a :class:`SecurityReview`."""
   253→        findings: list[SecurityFindingDetail] = []
   254→        findings.extend(self._run_custom_rules(repository, workspace))
   255→        findings.extend(self._run_bandit(repository, workspace))
   256→        findings.extend(self._run_detect_secrets(repository, workspace))
   257→        # Deduplicate by (file, line, category).
   258→        findings = self._deduplicate(findings)
   259→        severity = self._overall_severity(findings)
   260→        score = self._compute_score(findings)
   261→        review = SecurityReview(
   262→            findings=findings,
   263→            overall_severity=severity,
   264→            security_score=score,
   265→            summary=self._build_summary(findings, score),
   266→            owasp_top10_coverage=self._owasp_coverage(findings),
   267→            credential_exposure_count=sum(
   268→                1 for f in findings if "hardcoded" in f.category or "key" in f.category
   269→            ),
   270→            injection_risk_count=sum(
   271→                1 for f in findings if "injection" in f.category or "eval" in f.category
   272→            ),
   273→            misconfiguration_count=sum(
   274→                1 for f in findings if f.category in {"debug_mode", "shell_true", "yaml_load"}
   275→            ),
   276→        )
   277→        return review
   278→
   279→    # ----- custom regex rules --------------------------------------------------
   280→
   281→    def _run_custom_rules(
   282→        self, repository: Repository, workspace: Path
   283→    ) -> list[SecurityFindingDetail]:
   284→        """Run all custom regex rules across source files."""
   285→        findings: list[SecurityFindingDetail] = []
   286→        for path in self._iter_source_files(workspace):
   287→            try:
   288→                content = path.read_text(encoding="utf-8", errors="ignore")
   289→            except OSError:
   290→                continue
   291→            rel = str(path.relative_to(workspace))
   292→            for rule in _CUSTOM_RULES:
   293→                # Skip .env-style rules unless the filename matches.
   294→                filename_pat = rule.get("filename_pattern")
   295→                if filename_pat and not re.search(filename_pat, path.name):
   296→                    continue
   297→                findings.extend(self._apply_rule(rule, rel, content))
   298→        return findings
   299→
   300→    def _apply_rule(
   301→        self, rule: dict[str, Any], file_path: str, content: str
   302→    ) -> list[SecurityFindingDetail]:
   303→        """Apply a single regex rule to a file's content."""
   304→        results: list[SecurityFindingDetail] = []
   305→        pattern = rule["pattern"]
   306→        try:
   307→            compiled = re.compile(pattern)
   308→        except re.error:
   309→            return results
   310→        for match in compiled.finditer(content):
   311→            line = content[: match.start()].count("\n") + 1
   312→            snippet = self._snippet(content, line)
   313→            results.append(
   314→                SecurityFindingDetail(
   315→                    title=rule["title"],
   316→                    category=rule["category"],
   317→                    severity=rule["severity"],
   318→                    risk_level=rule["severity"],
   319→                    cvss_estimate=self._cvss(rule["severity"]),
   320→                    file=file_path,
   321→                    line=line,
   322→                    code_snippet=snippet,
   323→                    why_risky=rule["why"],
   324→                    real_world_risk=rule["risk"],
   325→                    solution=rule["solution"],
   326→                    safe_code_example=rule["safe"],
   327→                    references=rule["refs"],
   328→                    tool="custom",
   329→                )
   330→            )
   331→        return results
   332→
   333→    @staticmethod
   334→    def _snippet(content: str, line: int, context: int = 1) -> str:
   335→        """Return a small code snippet around ``line``."""
   336→        lines = content.splitlines()
   337→        start = max(0, line - 1 - context)
   338→        end = min(len(lines), line + context)
   339→        return "\n".join(lines[start:end])
   340→
   341→    # ----- bandit --------------------------------------------------------------
   342→
   343→    def _run_bandit(self, repository: Repository, workspace: Path) -> list[SecurityFindingDetail]:
   344→        """Run Bandit if installed and collect findings."""
   345→        if self._skip_bandit:
   346→            return []
   347→        try:
   348→            proc = subprocess.run(
   349→                ["bandit", "-r", str(workspace), "-f", "json", "-q"],
   350→                capture_output=True,
   351→                text=True,
   352→                timeout=120,
   353→                check=False,
   354→            )
   355→        except (FileNotFoundError, subprocess.TimeoutExpired) as exc:
   356→            _logger.debug("Bandit unavailable: %s", exc)
   357→            return []
   358→        if proc.returncode not in (0, 1):
   359→            return []
   360→        try:
   361→            data = json.loads(proc.stdout or "{}")
   362→        except json.JSONDecodeError:
   363→            return []
   364→        findings: list[SecurityFindingDetail] = []
   365→        for item in data.get("results", []):
   366→            severity = self._bandit_severity(item.get("issue_severity", "MEDIUM"))
   367→            findings.append(
   368→                SecurityFindingDetail(
   369→                    title=item.get("test_name", "Bandit finding"),
   370→                    category=item.get("test_id", "bandit").lower(),
   371→                    severity=severity,
   372→                    risk_level=severity,
   373→                    cvss_estimate=self._cvss(severity),
   374→                    file=item.get("filename", ""),
   375→                    line=item.get("line_number"),
   376→                    code_snippet=item.get("issue_text", ""),
   377→                    why_risky=item.get("issue_text", ""),
   378→                    real_world_risk=self._bandit_risk(item.get("test_id", "")),
   379→                    solution=self._bandit_solution(item.get("test_id", "")),
   380→                    safe_code_example=None,
   381→                    references=[item.get("test_id", ""), item.get("more_info", "")],
   382→                    tool="bandit",
   383→                    confidence=self._bandit_confidence(item.get("issue_confidence", "MEDIUM")),
   384→                )
   385→            )
   386→        return findings
   387→
   388→    @staticmethod
   389→    def _bandit_severity(label: str) -> RiskLevel:
   390→        return {
   391→            "HIGH": RiskLevel.HIGH,
   392→            "MEDIUM": RiskLevel.MEDIUM,
   393→            "LOW": RiskLevel.LOW,
   394→        }.get(label.upper(), RiskLevel.MEDIUM)
   395→
   396→    @staticmethod
   397→    def _bandit_confidence(label: str) -> float:
   398→        return {"HIGH": 0.9, "MEDIUM": 0.6, "LOW": 0.3}.get(label.upper(), 0.5)
   399→
   400→    @staticmethod
   401→    def _bandit_risk(test_id: str) -> str:
   402→        return {
   403→            "B101": "assert statements are stripped in optimized mode, hiding security checks.",
   404→            "B301": "Pickle deserialization can execute arbitrary code.",
   405→            "B602": "subprocess with shell=True enables command injection.",
   406→            "B608": "String SQL construction enables SQL injection.",
   407→        }.get(test_id, "This pattern is a known security anti-pattern.")
   408→
   409→    @staticmethod
   410→    def _bandit_solution(test_id: str) -> str:
   411→        return {
   412→            "B101": "Remove assert for security checks; use a proper if/raise.",
   413→            "B301": "Use a safe serialization format like JSON.",
   414→            "B602": "Pass arguments as a list with shell=False.",
   415→            "B608": "Use parameterized queries.",
   416→        }.get(test_id, "Follow the Bandit recommendation in the linked docs.")
   417→
   418→    # ----- detect-secrets ------------------------------------------------------
   419→
   420→    def _run_detect_secrets(
   421→        self, repository: Repository, workspace: Path
   422→    ) -> list[SecurityFindingDetail]:
   423→        """Run detect-secrets if installed and collect findings."""
   424→        if self._skip_detect_secrets:
   425→            return []
   426→        try:
   427→            proc = subprocess.run(
   428→                ["detect-secrets", "scan", str(workspace)],
   429→                capture_output=True,
   430→                text=True,
   431→                timeout=120,
   432→                check=False,
   433→            )
   434→        except (FileNotFoundError, subprocess.TimeoutExpired) as exc:
   435→            _logger.debug("detect-secrets unavailable: %s", exc)
   436→            return []
   437→        if proc.returncode != 0:
   438→            return []
   439→        try:
   440→            data = json.loads(proc.stdout or "{}")
   441→        except json.JSONDecodeError:
   442→            return []
   443→        findings: list[SecurityFindingDetail] = []
   444→        for file_path, entries in data.get("results", {}).items():
   445→            for entry in entries:
   446→                severity = self._secret_severity(entry.get("type", ""))
   447→                findings.append(
   448→                    SecurityFindingDetail(
   449→                        title=f"Secret detected: {entry.get('type', 'unknown')}",
   450→                        category=entry.get("type", "secret").lower(),
   451→                        severity=severity,
   452→                        risk_level=severity,
   453→                        cvss_estimate=self._cvss(severity),
   454→                        file=file_path,
   455→                        line=entry.get("line_number"),
   456→                        code_snippet=None,
   457→                        why_risky="A live secret was detected in source control.",
   458→                        real_world_risk="The secret grants access to the associated service and must be rotated.",
   459→                        solution="Rotate the secret immediately and load it from a secret manager.",
   460→                        safe_code_example="secret = os.environ['SECRET_NAME']",
   461→                        references=[entry.get("type", "")],
   462→                        tool="detect-secrets",
   463→                    )
   464→                )
   465→        return findings
   466→
   467→    @staticmethod
   468→    def _secret_severity(secret_type: str) -> RiskLevel:
   469→        high = {"AWS Access Key", "Stripe", "GitHub Token", "JWT Token"}
   470→        if any(h in secret_type for h in high):
   471→            return RiskLevel.CRITICAL
   472→        if "Key" in secret_type or "Token" in secret_type:
   473→            return RiskLevel.HIGH
   474→        return RiskLevel.MEDIUM

   476→    # ----- helpers -------------------------------------------------------------
   477→
   478→    def _iter_source_files(self, workspace: Path):  # type: ignore[no-untyped-def]
   479→        skip_dirs = {".git", "node_modules", ".venv", "venv", "__pycache__", "dist", "build"}
   480→        for root, dirs, files in os.walk(workspace):
   481→            dirs[:] = [d for d in dirs if d not in skip_dirs]
   482→            for name in files:
   483→                if name.endswith(
   484→                    (
   485→                        ".py",
   486→                        ".js",
   487→                        ".ts",
   488→                        ".jsx",
   489→                        ".tsx",

   526→    @staticmethod
   527→    def _compute_score(findings: Sequence[SecurityFindingDetail]) -> float:
   528→        """Compute a 0-100 security score from findings."""
   529→        penalties = {
   530→            RiskLevel.CRITICAL: 30,
   531→            RiskLevel.HIGH: 15,
   532→            RiskLevel.MEDIUM: 7,
   533→            RiskLevel.LOW: 3,
   534→            RiskLevel.INFO: 0,
   535→        }
   536→        score = 100.0
   537→        for finding in findings:
   538→            score -= penalties.get(finding.severity, 0)
   539→        return max(0.0, score)

   573→    @staticmethod
   574→    def _build_summary(findings: Sequence[SecurityFindingDetail], score: float) -> str:
   575→        """Build a one-paragraph engineering summary."""
   576→        if not findings:
   577→            return "No security findings were detected by the configured scanners."
   578→        critical = sum(1 for f in findings if f.severity == RiskLevel.CRITICAL)
   579→        high = sum(1 for f in findings if f.severity == RiskLevel.HIGH)
   580→        medium = sum(1 for f in findings if f.severity == RiskLevel.MEDIUM)
   581→        parts = [
   582→            f"Security scan identified {len(findings)} finding(s)",
   583→            f"({critical} critical, {high} high, {medium} medium)",
   584→            f"with an overall security score of {score:.0f}/100.",
   585→        ]
   586→        if critical > 0:
   587→            parts.append("Critical findings require immediate remediation and secret rotation.")
   588→        elif high > 0:
   589→            parts.append("High-severity findings should be addressed before the next release.")
   590→        else:
   591→            parts.append(
   592→                "The repository has a moderate security posture with room for improvement."
   593→            )
   594→        return " ".join(parts)
